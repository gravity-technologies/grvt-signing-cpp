  list(APPEND test_names signing)
endif()

  add_executable(test_${name} test_${name}.cpp $<TARGET_OBJECTS:grvt_sdk_test_main>)
  target_link_libraries(test_${name} PRIVATE grvt::signing grvt_sdk_flags)
  if(GRVT_SIGNING_WITH_SECP256K1 AND (name STREQUAL "signing" OR name STREQUAL "alloc"))
    target_link_libraries(test_${name} PRIVATE grvt::signer)
    target_compile_definitions(test_${name} PRIVATE GRVT_WITH_SIGNER=1)
  endif()
  target_include_directories(test_${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../src)
  target_compile_definitions(test_${name} PRIVATE GRVT_SIGNING_CASES_FILE="${GRVT_SIGNING_CASES_FILE}")
  add_test(NAME ${name} COMMAND test_${name})
endforeach()
